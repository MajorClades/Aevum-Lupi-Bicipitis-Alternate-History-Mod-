custom_button = { 
	name = experiment_1_button
	potential = {
		has_country_flag = mech_ongoing_experiment
		NOT = {
			has_country_flag = complete_1_experiment
		}
	}
	trigger = {
		if = {
			limit = {
				has_country_flag = experiment_demand_1a_flag
			}
			
			ROOT = {
				any_owned_province = {
					has_building = workshop
				}
			}
		}
		if = {
			limit = {
				has_country_flag = experiment_demand_1b_flag
			}
			
			ROOT = {
				natural_scientist = 2
			}
		}
		if = {
			limit = {
				has_country_flag = experiment_demand_1c_flag
			}
			
			any_province = {
				num_of_units_in_province = {
					amount = 5
					who = ROOT
					type = infantry
				}
				OR = {
					trade_goods = copper
					trade_goods = iron
				}
			}
		}
		if = {
			limit = {
				has_country_flag = experiment_demand_1d_flag
			}
			
			ROOT = {
				artist = 2
			}
		}
		if = {
			limit = {
				has_country_flag = experiment_demand_1e_flag
			}
			
			ROOT = {
				theologian = 2
			}
		}
		if = {
			limit = {
				has_country_flag = experiment_demand_1f_flag
			}
			
			any_province = {
				num_of_units_in_province = {
					amount = 5
					who = ROOT
					type = infantry
				}
				OR = {
					trade_goods = gold
					trade_goods = gems
				}
			}
		}
		if = {
			limit = {
				has_country_flag = experiment_demand_1g_flag
			}
			
			is_at_war = yes
		}
	}
	effect = {	
		if = {
			limit = {
				has_country_flag = experiment_demand_1a_flag
			}
			random_owned_province = {
					limit = { has_building = workshop }
					remove_building = workshop
				}
				add_years_of_income = 0.15
		}
		if = {
			limit = {
				has_country_flag = experiment_demand_1b_flag
			}
			kill_advisor = natural_scientist
		}
		if = {
			limit = {
				has_country_flag = experiment_demand_1c_flag
			}
			add_manpower = -1
		}
		if = {
			limit = {
				has_country_flag = experiment_demand_1d_flag
			}
			kill_advisor = artist 
			add_adm_power = 100
		}
		if = {
			limit = {
				has_country_flag = experiment_demand_1e_flag
			}
			kill_advisor = theologian
			add_years_of_income = 0.1
		}
		if = {
			limit = {
				has_country_flag = experiment_demand_1f_flag
			}
			add_manpower = -1
			add_years_of_income = 0.15
		}
		if = {
			limit = {
				has_country_flag = experiment_demand_1g_flag
			}
			add_war_exhaustion = 1
		}
		set_country_flag = complete_1_experiment
		clr_country_flag = experiment_demand_1a_flag
		clr_country_flag = experiment_demand_1b_flag
		clr_country_flag = experiment_demand_1c_flag

custom_button = {
	name = railway_construction_button
	potential = {
		FROM = {
			has_country_flag = railway_construction_button_flag
		}
		owned_by = FROM
		NOT = {
			has_province_modifier = railway_in_construction
		}
		NOT = {
			has_province_modifier = mekh_railway
		}
	}
	trigger = {
		owner = {
				years_of_income = 0.5
			}
		is_core = owner
	}
	effect = {
		owner = {
				add_years_of_income = -0.5
			}
		add_province_modifier = {
			name = mekh_railway
			duration = -1
		}
		add_province_modifier = {
			name = railway_in_construction
			duration = 3650
		}
	}
	tooltip = railway_construction_button_tt
