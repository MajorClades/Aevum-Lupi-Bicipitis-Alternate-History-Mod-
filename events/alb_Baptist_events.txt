country_event = { #celebration event council of the successors
    id = bapti.1
    title = bapti.1.t
    desc = bapti.1.d
    picture = BIG_ARABIAN_CITY_eventPicture
    is_triggered_only = yes
    option = {
        name = bapti.1.a
        change_variable = {
					which = baptist_voting_A
					value = 2
				}
        set_country_flag = bap_voted_A_flag
        every_subject_country = {
				limit = {
					is_subject_of_type = iyali_vassal
				}
				country_event = {
					id = bapti.2
					days = 1
				}
			}
        country_event = { #event 1 aspect
				id = bapti.3
				days = 25
			}

    }
     option = {
        name = bapti.1.b
        change_variable = {
					which = baptist_voting_B
					value = 2
				}
        set_country_flag = bap_voted_B_flag
        every_subject_country = {
				limit = {
					religion = baptistism
				}
				country_event = {
					id = bapti.2
					days = 1
				}
			}
       if = {
            limit = {
                OR = {
                    has_country_flag = banned_baptist_1_aspect_A
                    has_country_flag = banned_baptist_1_aspect_B
                }
            }
            if = {
                limit = {
                    OR = {
                        has_country_flag = banned_baptist_2_aspect_A
                        has_country_flag = banned_baptist_2_aspect_B
                    }
                }
                country_event = { #event 3 aspect
                    id = bapti.5
                    days = 25
                }
            }else = {
                country_event = { #event 2 aspect
                    id = bapti.4
                    days = 25
                }
            }
       }else = {
            country_event = { #event 1 aspect
                    id = bapti.3
                    days = 25
                }
       }
    }
}
