country_event = { #1 aspect
    id = bapti.3
    title = bapti.3.t
    desc = bapti.3.d
    picture = BIG_ARABIAN_CITY_eventPicture
    is_triggered_only = yes
    immediate = {
        hidden_effect = {
            clr_country_flag = bap_voted_A_flag
            clr_country_flag = bap_voted_B_flag
        }
    }
    option = {
        name = bapti.3.a
        if = {
            limit = {
                check_variable = {
                    which = baptist_voting_A
                    which = baptist_voting_B
                }
            }
            set_country_flag = unlocked_baptist_1_aspect_A
            tooltip = {
                add_church_aspect = unlockable_1_A_aspect
            }
        }else = {
            set_country_flag = unlocked_baptist_1_aspect_B
            tooltip = {
                add_church_aspect = unlockable_1_B_aspect
            }
        }
    }
