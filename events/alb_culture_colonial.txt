country_event = {
    id = alb_culture_colonial.1
    title = "alb_culture_colonial.title"
    desc = { #RAUSULIANO
        trigger = {
            culture_group = latin
            NOT = { primary_culture = rausuliano }
			NOT = { primary_culture = iocotanobo }
            any_owned_province = {
                contrinent = north_america
                culture_group = latin
                NOT = { culture = iocotanobo }
            }
        }
        desc = alb_culture_colonial.lat.rau.desc
    }
    picture = COLONIZATION_eventPicture
    is_triggered_only = yes
    trigger = {
        capital_scope = {
			OR = {
				continent = north_america
				continent = south_america
			}
		}
		NOT = { has_country_flag = rejected_new_colonial_culture }
        OR = {
            #ELIGIBLE RAUSULIANO
            AND = { 
                culture_group = latin
                NOT = { primary_culture = rausuliano }
			    NOT = { primary_culture = iocotanobo }
                any_owned_province = {
                    contrinent = north_america
                    culture_group = latin
                    NOT = { culture = iocotanobo }
                }
            }
        }
    }
    mean_time_to_happen = {
		days = 1
	}
    immediate = {
        hidden_effect = {
            if = { #RAUSULIANO
                limit = {
                    culture_group = latin
                    NOT = { primary_culture = rausuliano }
			        NOT = { primary_culture = iocotanobo }
                    any_owned_province = {
                        contrinent = north_america
                        culture_group = latin
                        NOT = { culture = iocotanobo }
                    }
                }
                random_owned_province = {
                    limit = {
                        contrinent = north_america
                        culture_group = latin
                        NOT = { culture = iocotanobo }
                    }
                    save_event_target_as = rausuliano_province
                }
            }
        }
    }
    option = { #BECOME RAUSULIANO
        trigger = {
            culture_group = latin
            NOT = { primary_culture = iocotanobo }
            any_owned_province = {
                contrinent = north_america
                culture_group = latin
                NOT = { culture = iocotanobo }
            }
        }
        name = "alb_culture_colonial.lat.rau.a"
        if = {
            limit = {
                NOT = {
                    primary_culture = rausuliano
                }
            }
            change_primary_culture = rausuliano
        }
        event_target:rausuliano_province = {
            area = {
                limit = {
                    owned_by = ROOT
                    culture_group = latin
                    NOT = { culture = iocotanobo }
                }
                change_culture = rausuliano
            }
        }
        set_country_flag = embraced_new_colonial_culture
    }
    option = { #REJECT RAUSULIANO
        trigger = {
            culture_group = latin
            any_owned_province = {
                contrinent = north_america
                culture_group = latin
                NOT = { culture = iocotanobo }
            }
            ai = no
			NOT = { has_country_flag = embraced_new_colonial_culture }
			NOT = { primary_culture = rausuliano }
        }
        name = "alb_culture_colonial.lat.rau.b
    }
}
